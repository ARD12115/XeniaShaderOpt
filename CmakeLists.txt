# =============================================================================
#  Compiler warning / optimisation helpers
# =============================================================================

function(xso_target_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE
            -Wall -Wextra -Wpedantic -Wno-unused-parameter)
    endif()
endfunction()

function(xso_target_optimise target)
    if(NOT MSVC)
        target_compile_options(${target} PRIVATE
            $<$<CONFIG:Release>:        -O3 -DNDEBUG>
            $<$<CONFIG:RelWithDebInfo>: -O2 -g -DNDEBUG>
            $<$<CONFIG:Debug>:          -O0 -g>
        )
    else()
        target_compile_options(${target} PRIVATE
            $<$<CONFIG:Release>:        /O2 /DNDEBUG>
            $<$<CONFIG:Debug>:          /Od /Zi>
        )
    endif()
endfunction()

)

    $<$<COMPILE_LANGUAGE:CUDA>:
        --use_fast_math
        --generate-line-info
        -Xcompiler -Wall
        $<$<CONFIG:Release>:        -O3>
        $<$<CONFIG:RelWithDebInfo>: -O2 -g -G>
        $<$<CONFIG:Debug>:          -O0 -g -G --device-debug>
    >
)

    PUBLIC  CUDA::cudart

)

    macro(xso_add_test test_name src_file link_libs)
        add_executable(${test_name} ${src_file})
        target_link_libraries(${test_name} PRIVATE ${link_libs})
        target_include_directories(${test_name} PRIVATE
            ${CMAKE_CURRENT_SOURCE_DIR}/include
            ${CMAKE_CURRENT_SOURCE_DIR}/cuda
        )
        target_compile_definitions(${test_name} PRIVATE ${XSO_DEFINES})
        if(XSO_SANITIZER_FLAGS)
            target_compile_options(${test_name} PRIVATE ${XSO_SANITIZER_FLAGS})
            target_link_options(${test_name}    PRIVATE ${XSO_SANITIZER_FLAGS})
        endif()
        add_test(NAME ${test_name} COMMAND ${test_name})
    endmacro()

    # Core cache correctness
